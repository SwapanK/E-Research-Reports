// ---------- helpers ----------
function iconCircle(slide, x, y, d, bg, iconName, iconColor) {
  slide.addShape('ellipse', { x, y, w: d, h: d, fill: { color: bg }, line: { type: 'none' } });
  const pad = d * 0.26;
  slide.addImage({ path: ICON(iconName, iconColor), x: x + pad, y: y + pad, w: d - pad * 2, h: d - pad * 2 });
}

function pageHeader(slide, kicker, title, iconName) {
  iconCircle(slide, 0.5, 0.45, 0.62, BLUE, iconName, 'white');
  slide.addText(kicker.toUpperCase(), {
    x: 1.28, y: 0.42, w: 8, h: 0.28, fontFace: FONT_BODY, fontSize: 12, color: BLUE, bold: true, charSpacing: 2,
  });
  slide.addText(title, {
    x: 1.28, y: 0.66, w: 10.5, h: 0.55, fontFace: FONT_HEAD, fontSize: 27, color: NAVY, bold: true,
  });
  slide.addShape('line', { x: 0.5, y: 1.42, w: PW - 1.0, h: 0, line: { color: 'DDE6EC', width: 1.25 } });
}

function footer(slide, pageNum) {
  slide.addText('Vulnerability Module — User Guide', {
    x: 0.5, y: PH - 0.42, w: 7, h: 0.3, fontFace: FONT_BODY, fontSize: 9, color: MUTED,

    x: PW - 1.0, y: PH - 0.42, w: 0.5, h: 0.3, fontFace: FONT_BODY, fontSize: 9, color: MUTED, align: 'right',
  });
}

function bgFill(slide, color) {
  slide.background = { color };
}

function moduleSlide({ pageNum, section, title, icon, purpose, steps, right }) {
  const s = pres.addSlide();
  bgFill(s, WHITE);
  pageHeader(s, section, title, icon);

  s.addShape('roundRect', { x: 0.5, y: 1.6, w: 12.3, h: 0.62, rectRadius: 0.06, fill: { color: CARD }, line: { type: 'none' } });
  s.addText([{ text: 'PURPOSE   ', options: { bold: true, color: BLUE, fontSize: 11 } }, { text: purpose, options: { color: TEXT, fontSize: 11.5 } }], {
    x: 0.7, y: 1.6, w: 11.9, h: 0.62, fontFace: FONT_BODY, valign: 'middle', lineSpacingMultiple: 1.05,
  });

  // Left column: steps
  s.addText('LEFT PANEL — STEP BY STEP', { x: 0.5, y: 2.42, w: 7.4, h: 0.28, fontFace: FONT_BODY, fontSize: 11, bold: true, color: BLUE, charSpacing: 1 });

  let y = 2.78;
  steps.forEach((st, i) => {
    iconCircle(s, 0.5, y, 0.4, BLUE, 'check', 'white');
    s.addText(st.t, { x: 1.05, y: y - 0.03, w: 6.9, h: 0.3, fontFace: FONT_BODY, fontSize: 12.5, bold: true, color: NAVY });
    s.addText(st.d, { x: 1.05, y: y + 0.27, w: 6.9, h: st.h || 0.55, fontFace: FONT_BODY, fontSize: 10, color: MUTED, lineSpacingMultiple: 1.05 });
    y += (st.h || 0.55) + 0.42;
  });

  // Right column: tabs / notes card
  s.addShape('roundRect', { x: 8.2, y: 2.42, w: 4.6, h: y - 2.42 > 4.35 ? y - 2.42 : 4.35, rectRadius: 0.06, fill: { color: NAVY }, line: { type: 'none' } });
  s.addText(right.heading, { x: 8.45, y: 2.62, w: 4.1, h: 0.3, fontFace: FONT_BODY, fontSize: 11, bold: true, color: LBLUE, charSpacing: 1 });
  let ry = 3.0;
  right.items.forEach((it) => {
    s.addText('•  ' + it, { x: 8.45, y: ry, w: 4.1, h: 0.5, fontFace: FONT_BODY, fontSize: 10, color: 'E3EDF3', lineSpacingMultiple: 1.1 });
    ry += 0.5;
  });

  footer(s, pageNum);
  return s;
}
